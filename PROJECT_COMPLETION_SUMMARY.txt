🎯 ENDPOINTS API
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

POST /api/auth/register
  Request:  {username, email, password, role}
  Response: 201 CREATED {id, username, email, role, token: null}
  Errors:   409 CONFLICT (email existe), 400 BAD REQUEST (validation)

POST /api/auth/login
  Request:  {email, password}
  Response: 200 OK {id, username, email, role, token}
  Errors:   401 UNAUTHORIZED (credentials), 400 BAD REQUEST (validation)

GET /api/protected/resource
  Header:   Authorization: Bearer <token>
  Response: 200 OK (data protégée)
  Errors:   401 UNAUTHORIZED (token invalide/expiré)
